cmake_minimum_required(VERSION 3.8)
project(icm20948_imu)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
add_executable(icm20948_node src/icm20948_node.cpp)
target_compile_features(icm20948_node PUBLIC cxx_std_17)
ament_target_dependencies(icm20948_node rclcpp sensor_msgs)
install(TARGETS icm20948_node DESTINATION lib/${PROJECT_NAME})
ament_package()
